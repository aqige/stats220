# load magick library
library(magick)

#read image
img <-image_read(path = "https://www.easetext.com/assets/img/tutorial/jpg-to-text.jpg")
#show original image size
image_info(img)


# -------------------------
# Part D: Create static meme
# -------------------------

#rescale image size to 600*450,then add red text png as image result format.
meme <- img %>%
  image_scale("600x450")%>%
  image_annotate(
    ".png",
    size = 60,
    color = "red",
    location = "+450+350"
  )

#save image as my_meme.png
meme %>%
  image_write("my_meme.png")

# -------------------------
# Part E: Adding animation 
# -------------------------

# Create frame 1
frame1 <- img %>%
  image_scale("600x450")%>%
  image_annotate(
    ".png",
    size = 40,
    color = "white",
    location = "+450+350"
  )

# Create frame 2
frame2 <- img %>%
  image_scale("600x450")%>%
  image_annotate(
    ".png",
    size = 50,
    color = "red",
    location = "+450+350"
  )

# Create frame 3
frame3 <- img %>%
  image_scale("600x450")%>%
  image_annotate(
    ".png",
    size = 60,
    color = "blue",
    location = "+450+350"
  )

# Create frame 4
frame4 <- img %>%
  image_scale("600x450")%>%
  image_annotate(
    ".png",
    size = 70,
    color = "green",
    location = "+450+350"
  )

# Combine frames into one animation object
frames <- c(frame1, frame2, frame3, frame4)

# Animate the frames
animation <- frames %>%
  image_animate(fps = 1)

# Save the animated GIF
animation %>%
  image_write("my_animated_meme.gif")
